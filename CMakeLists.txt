cmake_minimum_required(VERSION 3.25)
project(Wren VERSION 0.2.0 DESCRIPTION "Electronic structure and kinetic Monte Carlo" LANGUAGES NONE)
option(WREN_BUILD_CPP "Build the C++ KMC implementation" ON)
option(WREN_BUILD_FORTRAN "Build the Fortran KMC implementation" ON)
option(WREN_BUILD_RUST "Build the Rust KMC implementation with Cargo" ON)
option(WREN_BUILD_PYTHON "Build the Python wheel with uv and install its CLI commands" ON)
if(WREN_BUILD_CPP)
  enable_language(CXX)
endif()
if(WREN_BUILD_FORTRAN)
  enable_language(Fortran)
endif()
# GNUInstallDirs normally derives lib/lib64 from a configured native ABI.
if(NOT WREN_BUILD_CPP AND NOT WREN_BUILD_FORTRAN)
  set(CMAKE_INSTALL_LIBDIR "lib" CACHE PATH "Library installation directory")
endif()
include(GNUInstallDirs)
include(CTest)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(CMAKE_Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${CMAKE_Fortran_MODULE_DIRECTORY}")
function(wren_program name source library)
  add_executable(${name} "${source}")
  target_link_libraries(${name} PRIVATE ${library})
endfunction()

add_custom_target(wren_benchmarks)
if(WREN_BUILD_CPP)
  add_library(ukmc_cpp INTERFACE)
  target_compile_features(ukmc_cpp INTERFACE cxx_std_17)
  target_include_directories(ukmc_cpp INTERFACE
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/wren>")
  wren_program(ukmc-cpp cpp/src/main.cpp ukmc_cpp)
  wren_program(generic-cpp examples/generic.cpp ukmc_cpp)
  add_executable(kernel-cpp EXCLUDE_FROM_ALL benchmarks/kernel.cpp)
  target_link_libraries(kernel-cpp PRIVATE ukmc_cpp)
  add_dependencies(wren_benchmarks kernel-cpp ukmc-cpp)
  install(TARGETS ukmc-cpp ukmc_cpp EXPORT WrenTargets)
  install(DIRECTORY cpp/include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/wren")
  if(BUILD_TESTING)
    wren_program(test-cpp cpp/tests/test.cpp ukmc_cpp)
    add_test(NAME cpp COMMAND test-cpp)
    add_test(NAME generic_cpp COMMAND generic-cpp)
  endif()
endif()
if(WREN_BUILD_FORTRAN)
  add_library(ukmc_fortran STATIC fortran/ukmc.f90 fortran/ukmc_io.f90 fortran/ukmc_events.f90)
  target_include_directories(ukmc_fortran PUBLIC "$<BUILD_INTERFACE:${CMAKE_Fortran_MODULE_DIRECTORY}>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/wren>")
  if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
    target_compile_options(ukmc_fortran PRIVATE -std=f2008 "$<$<CONFIG:Debug>:-fcheck=all>")
  endif()
  wren_program(ukmc-fortran fortran/main.f90 ukmc_fortran)
  wren_program(generic-fortran examples/generic.f90 ukmc_fortran)
  add_executable(kernel-fortran EXCLUDE_FROM_ALL benchmarks/kernel.f90)
  target_link_libraries(kernel-fortran PRIVATE ukmc_fortran)
  add_dependencies(wren_benchmarks kernel-fortran ukmc-fortran)
  install(TARGETS ukmc-fortran ukmc_fortran EXPORT WrenTargets)
  install(DIRECTORY "${CMAKE_Fortran_MODULE_DIRECTORY}/" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/wren"
    FILES_MATCHING PATTERN "ukmc*.mod" PATTERN "UKMC*.mod" PATTERN "UKMC*.MOD")
  if(BUILD_TESTING)
    wren_program(test-fortran fortran/test.f90 ukmc_fortran)
    add_test(NAME fortran COMMAND test-fortran)
    add_test(NAME generic_fortran COMMAND generic-fortran)
  endif()
endif()
if(WREN_BUILD_CPP OR WREN_BUILD_FORTRAN)
  install(EXPORT WrenTargets FILE WrenConfig.cmake NAMESPACE Wren:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Wren")
  include(CMakePackageConfigHelpers)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WrenConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WrenConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Wren")
endif()

# For multi-config generators, CMake places native programs in bin/<config>.
if(CMAKE_CONFIGURATION_TYPES)
  set(wren_bin "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>")
else()
  set(wren_bin "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
endif()

if(WREN_BUILD_RUST)
  include(cmake/WrenRust.cmake)
  add_dependencies(wren_benchmarks wren_rust wren_rust_benchmark)
  if(BUILD_TESTING)
    add_test(NAME rust COMMAND "${CARGO_EXECUTABLE}" test ${cargo_flags} --profile "${cargo_profile}")
    add_test(NAME generic_rust COMMAND "${rust_example}")
  endif()
endif()
if(WREN_BUILD_CPP AND WREN_BUILD_RUST AND WREN_BUILD_FORTRAN)
  set(all_native ON)
endif()
if(WREN_BUILD_PYTHON OR (BUILD_TESTING AND all_native))
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
endif()
if(WREN_BUILD_PYTHON)
  include(cmake/WrenPython.cmake)
  if(BUILD_TESTING)
    add_test(NAME python_wheel_cli COMMAND "${Python3_EXECUTABLE}" -I
      "${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_wheel.py" "${wren_wheel}")
  endif()
endif()
if(BUILD_TESTING AND all_native)
  function(wren_python_test name)
    add_test(NAME ${name} COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests ${ARGN} -v)
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
      ENVIRONMENT "PYTHONPATH=${PROJECT_SOURCE_DIR}/src;WREN_BIN_DIR=${wren_bin}")
  endfunction()
  wren_python_test(four_language_conformance -p test_conformance.py)
  if(WREN_BUILD_PYTHON)
    wren_python_test(python_suite)
  endif()
endif()
