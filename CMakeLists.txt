cmake_minimum_required(VERSION 3.18)
project(universal_kmc LANGUAGES CXX Fortran)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_Fortran_MODULE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/modules)
add_library(ukmc_cpp INTERFACE)
target_include_directories(ukmc_cpp INTERFACE cpp/include)
add_executable(ukmc-cpp cpp/src/main.cpp)
target_link_libraries(ukmc-cpp PRIVATE ukmc_cpp)
add_library(ukmc_fortran fortran/ukmc.f90 fortran/ukmc_io.f90 fortran/ukmc_events.f90)
target_include_directories(ukmc_fortran PUBLIC ${CMAKE_Fortran_MODULE_DIRECTORY})
add_executable(ukmc-fortran fortran/main.f90)
target_link_libraries(ukmc-fortran PRIVATE ukmc_fortran)
enable_testing()
add_executable(test-cpp cpp/tests/test.cpp)
target_link_libraries(test-cpp PRIVATE ukmc_cpp)
add_test(NAME cpp COMMAND test-cpp)
add_executable(test-fortran fortran/test.f90)
target_link_libraries(test-fortran PRIVATE ukmc_fortran)
add_test(NAME fortran COMMAND test-fortran)

add_executable(generic-cpp examples/generic.cpp)
target_link_libraries(generic-cpp PRIVATE ukmc_cpp)
add_executable(generic-fortran examples/generic.f90)
target_link_libraries(generic-fortran PRIVATE ukmc_fortran)
add_test(NAME generic_cpp COMMAND generic-cpp)
add_test(NAME generic_fortran COMMAND generic-fortran)
